  <script>
    const STORAGE_KEY = "grantTrackerData_v1";

    function loadGrants() {
      const raw = localStorage.getItem(STORAGE_KEY);
      if (!raw) return [];
      try {
        return JSON.parse(raw);
      } catch {
        return [];
      }
    }

    function saveGrants(grants) {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(grants));
    }

    function generateId() {
      return "g_" + Date.now() + "_" + Math.random().toString(16).slice(2);
    }

    function getFormData() {
      return {
        id: document.getElementById("grantId").value || generateId(),
        grantName: document.getElementById("grantName").value.trim(),
        programName: document.getElementById("programName").value.trim(),
        grantSize: document.getElementById("grantSize").value.trim(),
        owner: document.getElementById("owner").value.trim(),
        status: document.getElementById("status").value,
        targetCompletion: document.getElementById("targetCompletion").value,
        applicationDue: document.getElementById("applicationDue").value,
        contactInfo: document.getElementById("contactInfo").value.trim(),
        notes: document.getElementById("notes").value.trim()
      };
    }

    function setFormData(grant) {
      document.getElementById("grantId").value = grant.id || "";
      document.getElementById("grantName").value = grant.grantName || "";
      document.getElementById("programName").value = grant.programName || "";
      document.getElementById("grantSize").value = grant.grantSize || "";
      document.getElementById("owner").value = grant.owner || "";
      document.getElementById("status").value = grant.status || "Research";
      document.getElementById("targetCompletion").value = grant.targetCompletion || "";
      document.getElementById("applicationDue").value = grant.applicationDue || "";
      document.getElementById("contactInfo").value = grant.contactInfo || "";
      document.getElementById("notes").value = grant.notes || "";
    }

    function clearForm() {
      setFormData({
        id: "",
        grantName: "",
        programName: "",
        grantSize: "",
        owner: "",
        status: "Research",
        targetCompletion: "",
        applicationDue: "",
        contactInfo: "",
        notes: ""
      });
    }

    function renderTable() {
      const grants = loadGrants();
      const tbody = document.getElementById("grantsTableBody");
      tbody.innerHTML = "";

      const filterStatus = document.getElementById("filterStatus").value;
      const filterOwner = document.getElementById("filterOwner").value.trim().toLowerCase();

      const filtered = grants.filter(g => {
        const statusMatch = !filterStatus || g.status === filterStatus;
        const ownerMatch = !filterOwner || (g.owner || "").toLowerCase().includes(filterOwner);
        return statusMatch && ownerMatch;
      });

      if (filtered.length === 0) {
        const tr = document.createElement("tr");
        const td = document.createElement("td");
        td.colSpan = 10;
        td.textContent = "No grants found. Add a grant using the form on the left.";
        td.style.fontStyle = "italic";
        tr.appendChild(td);
        tbody.appendChild(tr);
        return;
      }

      filtered.forEach(grant => {
        const tr = document.createElement("tr");

        const tdGrant = document.createElement("td");
        tdGrant.textContent = grant.grantName;
        tr.appendChild(tdGrant);

        const tdProgram = document.createElement("td");
        tdProgram.textContent = grant.programName || "";
        tr.appendChild(tdProgram);

        const tdSize = document.createElement("td");
        tdSize.textContent = grant.grantSize ? `\$${Number(grant.grantSize).toLocaleString()}` : "";
        tr.appendChild(tdSize);

        const tdStatus = document.createElement("td");
        const spanStatus = document.createElement("span");
        spanStatus.textContent = grant.status;
        spanStatus.className = "status-pill status-" + grant.status;
        tdStatus.appendChild(spanStatus);
        tr.appendChild(tdStatus);

        const tdOwner = document.createElement("td");
        tdOwner.textContent = grant.owner || "";
        tr.appendChild(tdOwner);

        const tdTarget = document.createElement("td");
        tdTarget.textContent = grant.targetCompletion || "";
        tr.appendChild(tdTarget);

        const tdDue = document.createElement("td");
        tdDue.textContent = grant.applicationDue || "";
        tr.appendChild(tdDue);

        const tdContact = document.createElement("td");
        tdContact.textContent = grant.contactInfo || "";
        tr.appendChild(tdContact);

        const tdNotes = document.createElement("td");
        tdNotes.textContent = grant.notes || "";
        tr.appendChild(tdNotes);

        const tdActions = document.createElement("td");
        tdActions.className = "actions";

        const editBtn = document.createElement("button");
        editBtn.textContent = "Edit";
        editBtn.className = "btn-secondary";
        editBtn.addEventListener("click", () => {
          setFormData(grant);
          window.scrollTo({ top: 0, behavior: "smooth" });
        });
        tdActions.appendChild(editBtn);

        const deleteBtn = document.createElement("button");
        deleteBtn.textContent = "Delete";
        deleteBtn.style.background = "#fed7d7";
        deleteBtn.style.color = "#742a2a";
        deleteBtn.addEventListener("click", () => {
          if (confirm("Delete this grant?")) {
            const updated = loadGrants().filter(g => g.id !== grant.id);
            saveGrants(updated);
            renderTable();
          }
        });
        tdActions.appendChild(deleteBtn);

        tr.appendChild(tdActions);

        tbody.appendChild(tr);
      });
    }

    document.getElementById("grantForm").addEventListener("submit", (e) => {
      e.preventDefault();
      const data = getFormData();
      if (!data.grantName) {
        alert("Grant name is required.");
        return;
      }

      const grants = loadGrants();
      const existingIndex = grants.findIndex(g => g.id === data.id);

      if (existingIndex >= 0) {
        grants[existingIndex] = data;
      } else {
        grants.push(data);
      }

      saveGrants(grants);
      clearForm();
      renderTable();
    });

    document.getElementById("resetFormBtn").addEventListener("click", () => {
      clearForm();
    });

    document.getElementById("filterStatus").addEventListener("change", renderTable);
    document.getElementById("filterOwner").addEventListener("input", renderTable);
    document.getElementById("clearFiltersBtn").addEventListener("click", () => {
      document.getElementById("filterStatus").value = "";
      document.getElementById("filterOwner").value = "";
      renderTable();
    });

    // Initial render
    renderTable();
  </script>
